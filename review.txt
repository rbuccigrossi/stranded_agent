87: def agent_for(session: Dict[str, Any], config: stranded.AgentConfig) -> Agent:
88:     """Return this session's live agent, rebuilding it when the configuration changes."""
89:     cached_config, cached_agent = AGENTS.get(session["id"], (None, None))
90:     if cached_agent is not None and cached_config == config:
91:         return cached_agent
92:     agent = stranded.build_agent(config, session.get("messages"), session.get("state"))
93:     AGENTS[session["id"]] = (config, agent)
94:     return agent

167: class StrandedHandler(BaseHTTPRequestHandler):
168:     server_version = "StrandedWeb/1.0"
169: 
170:     def log_message(self, format: str, *args: Any) -> None:
171:         return
172: 
173:     def send_json(self, value: Any, status: int = 200) -> None:
174:         body = json.dumps(value, default=str).encode("utf-8")
175:         self.send_response(status)
176:         self.send_header("Content-Type", "application/json; charset=utf-8")
177:         self.send_header("Content-Length", str(len(body)))
178:         self.end_headers()
179:         self.wfile.write(body)
180: 
181:     def read_json(self) -> Dict[str, Any]:
182:         length = int(self.headers.get("Content-Length", "0"))
183:         value = json.loads(self.rfile.read(length) or b"{}")
184:         return value if isinstance(value, dict) else {}
185: 
186:     def event(self, value: Dict[str, Any]) -> None:
187:         self.wfile.write(f"data: {json.dumps(value, default=str)}\n\n".encode("utf-8"))
188:         self.wfile.flush()
189: 
190:     def do_GET(self) -> None:
191:         path = urlparse(self.path).path
192:         if path == "/":
193:             body = HTML.encode("utf-8")
194:             self.send_response(200)
195:             self.send_header("Content-Type", "text/html; charset=utf-8")
196:             self.send_header("Content-Length", str(len(body)))
197:             self.end_headers()
198:             self.wfile.write(body)
199:         elif path == "/api/sessions":
200:             items = sorted(SESSIONS.values(), key=lambda item: item.get("updated", 0), reverse=True)
201:             self.send_json([session_summary(item) for item in items])
202:         elif path.startswith("/api/sessions/"):
203:             session = SESSIONS.get(path.rsplit("/", 1)[-1])
204:             self.send_json(session_view(session) if session else {"error": "session not found"},
205:                            200 if session else 404)
206:         else:
207:             self.send_json({"error": "not found"}, 404)
208: 
209:     def do_POST(self) -> None:
210:         path = urlparse(self.path).path
211:         try:
212:             if path == "/api/new":
213:                 self.send_json(session_summary(new_session()))
214:             elif path == "/api/chat":
215:                 self.chat(self.read_json())
216:             elif path == "/api/approve":
217:                 self.approve(self.read_json())
218:             else:
219:                 self.send_json({"error": "not found"}, 404)
220:         except (ValueError, json.JSONDecodeError) as error:
221:             self.send_json({"error": str(error)}, 400)
222: 
223:     def chat(self, payload: Dict[str, Any]) -> None:
224:         session = SESSIONS.get(payload.get("session_id")) or new_session()
225:         prompt = str(payload.get("prompt", "")).strip()
226:         if not prompt:
227:             self.send_json({"error": "prompt is required"}, 400)
228:             return
229:         if not session.get("messages"):
230:             session["label"] = prompt[:80]
231:         self.run_turn(session, config_from_payload(payload, session), prompt)
232: 
233:     def approve(self, payload: Dict[str, Any]) -> None:
234:         session = SESSIONS.get(payload.get("session_id"))
235:         interrupt_id = payload.get("interrupt_id")
236:         if not session or session["id"] not in AGENTS or not interrupt_id:
237:             self.send_json({"error": "approval request expired"}, 404)
238:             return
239:         self.run_turn(session, AGENTS[session["id"]][0],
240:                       stranded.resume_prompt(str(interrupt_id), bool(payload.get("approved"))))
241: 
242:     def run_turn(self, session: Dict[str, Any], config: stranded.AgentConfig, prompt: Any) -> None:
243:         """Stream one agent turn to the browser as server-sent events."""
244:         self.close_connection = True
245:         self.send_response(200)
246:         self.send_header("Content-Type", "text/event-stream; charset=utf-8")
247:         self.send_header("Cache-Control", "no-cache")
248:         self.send_header("Connection", "close")
249:         self.end_headers()
250:         self.event({"type": "session", "session_id": session["id"]})
251: 
252:         def emit(value: Dict[str, Any]) -> None:
253:             if value["type"] == "tool_call":
254:                 value = {**value, "detail": stranded.tool_detail(value["name"], value["arguments"])}
255:             self.event(value)
256: 
257:         agent = agent_for(session, config)
258:         turn = stranded.run_agent(agent, prompt, config, emit)
259:         record(session, agent)
260:         if turn.status == "approval_required":
261:             self.event({"type": "approval_required", "approval": turn.approval})
262:         self.event({"type": "done", "status": turn.status, "answer": turn.answer,
263:                     "reasoning": turn.reasoning, "usage": turn.usage, "error": turn.error})
